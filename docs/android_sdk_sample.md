# Android Telemetry Collector (Kotlin)

Drop the following class into an Android library module to capture telephony, carrier config, and network callbacks without shipping PII. The collector logs JSON telemetry that you can forward to your own instrumentation pipeline.

```kotlin
package com.example.netdiag

import android.content.Context
import android.net.ConnectivityManager
import android.net.Network
import android.net.NetworkCapabilities
import android.net.NetworkRequest
import android.telephony.CarrierConfigManager
import android.telephony.PhoneStateListener
import android.telephony.SignalStrength
import android.telephony.SubscriptionManager
import android.telephony.TelephonyManager
import org.json.JSONObject

class TelemetryCollector(private val ctx: Context) {
    private val tm = ctx.getSystemService(Context.TELEPHONY_SERVICE) as TelephonyManager
    private val sm = ctx.getSystemService(Context.TELEPHONY_SUBSCRIPTION_SERVICE) as SubscriptionManager
    private val cm = ctx.getSystemService(Context.CONNECTIVITY_SERVICE) as ConnectivityManager
    private val ccm = ctx.getSystemService(Context.CARRIER_CONFIG_SERVICE) as CarrierConfigManager

    private val listener = object : PhoneStateListener() {
        override fun onSignalStrengthsChanged(signalStrength: SignalStrength?) {
            signalStrength?.let {
                val json = JSONObject()
                json.put("rsrp_rssi", it.toString())
                flushTelemetry(json)
            }
        }

        override fun onDataConnectionStateChanged(state: Int, networkType: Int) {
            val j = JSONObject()
            j.put("dataState", state)
            j.put("networkType", networkType)
            flushTelemetry(j)
        }
    }

    fun start() {
        val subList = sm.activeSubscriptionInfoList
        val subId = if (!subList.isNullOrEmpty()) subList[0].subscriptionId else -1

        tm.listen(listener, PhoneStateListener.LISTEN_SIGNAL_STRENGTHS or PhoneStateListener.LISTEN_DATA_CONNECTION_STATE)

        val cfg = ccm.getConfigForSubId(subId)
        val jcfg = JSONObject()
        cfg?.keySet()?.forEach { k ->
            try { jcfg.put(k, cfg.get(k)) } catch (e: Exception) { }
        }
        flushTelemetry(JSONObject().put("carrierConfig", jcfg))

        val req = NetworkRequest.Builder().addTransportType(NetworkCapabilities.TRANSPORT_CELLULAR).build()
        cm.registerNetworkCallback(req, object : ConnectivityManager.NetworkCallback() {
            override fun onAvailable(network: Network) {
                flushTelemetry(JSONObject().put("network", "available"))
            }

            override fun onLost(network: Network) {
                flushTelemetry(JSONObject().put("network", "lost"))
            }
        })
    }

    private fun flushTelemetry(json: JSONObject) {
        // Replace with your own sink (local storage, HTTP POST, etc.)
        // Ensure you redact IMSI/ICCID/MSISDN before exporting telemetry off-device.
        android.util.Log.i("NetDiag", json.toString())
    }

    fun stop() {
        tm.listen(listener, PhoneStateListener.LISTEN_NONE)
    }
}
```

## Usage Notes

- Request `READ_PHONE_STATE` and `READ_PRECISE_PHONE_STATE` where required.
- On Android 10+ many telephony APIs are gated; design your UX around user consent dialogs.
- Pair this collector with the diagnostic bundle generated by `phreak.py` for end-to-end case handoffs.
